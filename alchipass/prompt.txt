##code Raspberry
import time
import serial
import threading
import requests
import RPi.GPIO as GPIO
from mfrc522 import SimpleMFRC522
from gpiozero import TonalBuzzer
from gpiozero.tones import Tone

# Configuration matériel
PORT_SERIE = '/dev/ttyACM0'  # À changer en /dev/ttyUSB0 si nécessaire
VITESSE_BAUDS = 9600

# Configuration du digicode (Mode BOARD requis par MFRC522)
ROW_PINS = [29, 31, 33, 35]
COL_PINS = [32, 36, 38, 40]

KEYMAP = [
    ['1', '2', '3', 'A'],
    ['4', '5', '6', 'B'],
    ['7', '8', '9', 'C'],
    ['*', '0', '#', 'D']
]

# URL du serveur Backend
URL_BASE = 'http://127.0.0.1:3000/api'

# Configuration du buzzer KXG1205
BUZZER_PIN = 17  # Pin GPIO en mode BCM (gpiozero utilise BCM)
buzzer = TonalBuzzer(BUZZER_PIN)

# Variables globales partagées pour la saisie du code
mot_de_passe_en_cours = ""
etape_validation = False  # True quand on est dans la phase OUI/NON/RETOUR
dernier_ouvrier = ""  # Stocke le nom de l'ouvrier pour la validation
saisie_mot_de_passe_active = False  # True quand l'utilisateur est en train de saisir son mot de passe
uid_casque_actuel = None  # Stocke l'UID du casque actuellement en cours de traitement

# Liste des étapes de validation EPI dans l'ordre
ETAPES_EPI = [
    {"oui": "CASQUE_OK", "non": "CASQUE_NOK", "libelle": "casque"},
    {"oui": "GILET_OK", "non": "GILET_NOK", "libelle": "gilet"},
    {"oui": "GANTS_OK", "non": "GANTS_NOK", "libelle": "gants"},
    {"oui": "CHAUSSURES_OK", "non": "CHAUSSURES_NOK", "libelle": "chaussures"}
]
etape_actuelle = 0  # Index de l'étape en cours (0-3)

# Dictionnaire pour suivre l'état de connexion de chaque UID
# True = connecté, False = déconnecté
etats_connexion = {}

def initialiser_materiel():
    """Initialise la liaison série et la configuration des broches GPIO."""
    GPIO.setmode(GPIO.BOARD)
    GPIO.setwarnings(False)

    # Configurer les lignes du digicode en entrée (Pull-Down)
    for row in ROW_PINS:
        GPIO.setup(row, GPIO.IN, pull_up_down=GPIO.PUD_DOWN)

    # Configurer les colonnes du digicode en sortie (LOW)
    for col in COL_PINS:
        GPIO.setup(col, GPIO.OUT)
        GPIO.output(col, GPIO.LOW)

    # Le buzzer est déjà initialisé avec gpiozero

    # Connexion Arduino
    try:
        ser = serial.Serial(PORT_SERIE, VITESSE_BAUDS, timeout=1)
        ser.flush()
        print(f"[MATÉRIEL] Connexion Arduino OK sur {PORT_SERIE}")
        return ser
    except serial.SerialException:
        print(f"[ERREUR] Impossible de trouver l'Arduino sur {PORT_SERIE}.")
        return None

def jouer_bip(duree, note="C5"):
    """Joue une note sur le buzzer pendant une durée spécifiée."""
    try:
        buzzer.play(note)
        time.sleep(duree)
        buzzer.stop()
    except Exception as e:
        print(f"[BUZZER] Erreur lors de la lecture du bip: {e}")

def jouer_bip_bon():
    """Joue un son 'bon' (2 notes aiguës joyeuses)."""
    print("[BUZZER] Son: BON ✅")
    jouer_bip(0.1, "A5")  # Note aiguë
    time.sleep(0.1)
    jouer_bip(0.1, "C6")  # Note plus aiguë

def jouer_bip_mauvais():
    """Joue un son 'mauvais' (bip grave et long)."""
    print("[BUZZER] Son: MAUVAIS ❌")
    jouer_bip(0.5, "G3")  # Note grave

def jouer_bip_erreur():
    """Joue un son d'erreur (descendant)."""
    print("[BUZZER] Son: ERREUR ⚠️")
    jouer_bip(0.15, "C5")
    time.sleep(0.05)
    jouer_bip(0.15, "G4")
    time.sleep(0.05)
    jouer_bip(0.3, "E4")

def envoyer_commande_arduino(arduino_serial, commande):
    """Envoie une commande brute textuelle à l'Arduino via le port série."""
    if arduino_serial and arduino_serial.is_open:
        try:
            arduino_serial.write(f"{commande}\n".encode('utf-8'))
            print(f"[SERIE] -> Envoi Arduino: {commande}")
        except Exception as e:
            print(f"[ERREUR] Échec de l'envoi série: {e}")

def lire_digicode():
    """Balaie la matrice du digicode pour détecter une touche unique."""
    pressed_key = None
    for col_num, col_pin in enumerate(COL_PINS):
        GPIO.output(col_pin, GPIO.HIGH)
        for row_num, row_pin in enumerate(ROW_PINS):
            if GPIO.input(row_pin) == GPIO.HIGH:
                pressed_key = KEYMAP[row_num][col_num]
                while GPIO.input(row_pin) == GPIO.HIGH:
                    time.sleep(0.01)
        GPIO.output(col_pin, GPIO.LOW)
    return pressed_key

def reinitialiser_validation():
    """Réinitialise complètement l'état de validation EPI."""
    global etape_validation, etape_actuelle, dernier_ouvrier, mot_de_passe_en_cours, saisie_mot_de_passe_active, uid_casque_actuel
    etape_validation = False
    etape_actuelle = 0
    dernier_ouvrier = ""
    mot_de_passe_en_cours = ""
    saisie_mot_de_passe_active = False
    uid_casque_actuel = None
    print("[VALIDATION] Réinitialisation complète - retour à l'écran d'attente")
    envoyer_commande_arduino(None, "RESET")

def est_connecte(uid):
    """Vérifie si l'UID est actuellement connecté."""
    return etats_connexion.get(uid, False)

def basculer_etat(uid):
    """Bascule l'état de connexion pour un UID donné."""
    etats_connexion[uid] = not etats_connexion.get(uid, False)
    return etats_connexion[uid]

def gerer_action_validation(arduino_serial, touche):
    """Gère les actions pendant la phase de validation EPI (OUI/NON/RETOUR)."""
    global etape_validation, etape_actuelle, dernier_ouvrier, uid_casque_actuel

    # Touche '1' = OUI
    if touche == '1':
        if etape_actuelle < len(ETAPES_EPI):
            commande = ETAPES_EPI[etape_actuelle]["oui"]
            print(f"[VALIDATION] OUI - {ETAPES_EPI[etape_actuelle]['libelle']} OK")
            envoyer_commande_arduino(arduino_serial, commande)

            # Jouer un bip bon pour chaque validation positive
            jouer_bip_bon()

            # Envoyer au backend
            try:
                requests.post(f"{URL_BASE}/evenement", json={
                    "action": commande,
                    "ouvrier": dernier_ouvrier,
                    "etape": etape_actuelle,
                    "uid_casque": uid_casque_actuel
                }, timeout=1)
            except requests.exceptions.RequestException:
                pass

            # Passer à l'étape suivante
            etape_actuelle += 1

            # Si toutes les étapes sont terminées, quitter le mode validation
            if etape_actuelle >= len(ETAPES_EPI):
                print("[VALIDATION] Toutes les vérifications EPI sont terminées")
                # Envoyer un message de fin de validation
                envoyer_commande_arduino(arduino_serial, "FIN_VALIDATION_OK")

                # CONNEXION DU CASQUE - Marquer comme connecté
                if uid_casque_actuel:
                    basculer_etat(uid_casque_actuel)
                    print(f"[RFID] CASQUE {uid_casque_actuel} CONNECTÉ - Validation EPI réussie")
                    # Jouer un son de succès final (3 bips rapides)
                    jouer_bip_bon()
                    time.sleep(0.1)
                    jouer_bip_bon()
                    time.sleep(0.1)
                    jouer_bip(0.2, "E6")

                    # Envoyer un événement de connexion du casque
                    try:
                        requests.post(f"{URL_BASE}/evenement", json={
                            "action": "CONNEXION_CASQUE_VALIDE",
                            "uid": uid_casque_actuel,
                            "ouvrier": dernier_ouvrier
                        }, timeout=1)
                    except:
                        pass

                # Réinitialiser complètement pour revenir à l'écran d'attente
                reinitialiser_validation()
        return

    # Touche '3' = NON
    if touche == '3':
        if etape_actuelle < len(ETAPES_EPI):
            commande = ETAPES_EPI[etape_actuelle]["non"]
            print(f"[VALIDATION] NON - {ETAPES_EPI[etape_actuelle]['libelle']} MANQUANT")
            envoyer_commande_arduino(arduino_serial, commande)

            # Jouer un bip mauvais pour chaque validation négative
            jouer_bip_mauvais()

            # Envoyer au backend
            try:
                requests.post(f"{URL_BASE}/evenement", json={
                    "action": commande,
                    "ouvrier": dernier_ouvrier,
                    "etape": etape_actuelle,
                    "uid_casque": uid_casque_actuel
                }, timeout=1)
            except requests.exceptions.RequestException:
                pass

            # Passer à l'étape suivante
            etape_actuelle += 1

            # Si toutes les étapes sont terminées, quitter le mode validation
            if etape_actuelle >= len(ETAPES_EPI):
                print("[VALIDATION] Toutes les vérifications EPI sont terminées")
                # Envoyer un message de fin de validation
                envoyer_commande_arduino(arduino_serial, "FIN_VALIDATION_NOK")

                # ÉCHEC DE CONNEXION - Ne pas connecter le casque
                if uid_casque_actuel:
                    print(f"[RFID] CASQUE {uid_casque_actuel} - Échec validation EPI, non connecté")
                    # Jouer un son d'échec prolongé
                    for _ in range(3):
                        jouer_bip_mauvais()
                        time.sleep(0.2)

                    try:
                        requests.post(f"{URL_BASE}/evenement", json={
                            "action": "ECHEC_VALIDATION_EPI",
                            "uid": uid_casque_actuel,
                            "ouvrier": dernier_ouvrier
                        }, timeout=1)
                    except:
                        pass

                # Réinitialiser complètement pour revenir à l'écran d'attente
                reinitialiser_validation()
        return

    # Touche '#' = RETOUR
    if touche == '#':
        if etape_actuelle > 0:
            # RETOUR : revenir à l'étape précédente
            etape_actuelle -= 1
            envoyer_commande_arduino(arduino_serial, "RETOUR")
            print(f"[VALIDATION] RETOUR - Retour à l'étape {etape_actuelle + 1}/{len(ETAPES_EPI)}")
            print(f"[VALIDATION] Vérifier le {ETAPES_EPI[etape_actuelle]['libelle']}")
            # Jouer un bip court pour indiquer le retour
            jouer_bip(0.05, "E5")
        else:
            # Si on est à la première étape, RETOUR annule la validation et RESET
            print("[VALIDATION] RETOUR - Annulation complète - retour à l'écran d'attente")
            envoyer_commande_arduino(arduino_serial, "RETOUR")  # D'abord RETOUR
            # Jouer un son d'annulation
            jouer_bip_erreur()
            reinitialiser_validation()  # Puis RESET complet
        return

    # Toute autre touche en mode validation est ignorée
    print(f"[VALIDATION] Touche ignorée en mode validation: {touche}")

def gerer_action_bouton(arduino_serial, touche):
    """Mappe les touches du digicode vers les commandes du cahier des charges."""
    global mot_de_passe_en_cours, etape_validation, saisie_mot_de_passe_active

    # Si on est en mode validation, gérer les touches OUI/NON/RETOUR
    if etape_validation:
        gerer_action_validation(arduino_serial, touche)
        return

    # 1. Traitement de la saisie d'un chiffre (0-9)
    if touche.isdigit():
        # Activer le mode saisie mot de passe
        saisie_mot_de_passe_active = True
        mot_de_passe_en_cours += touche
        # Jouer un bip court pour chaque chiffre saisi
        jouer_bip(0.03, "C5")
        print(f"[DIGICODE] Saisie en cours... ({mot_de_passe_en_cours})")
        return

    # 2. Touche '*' = effacer le dernier caractère (backspace)
    if touche == '*':
        if mot_de_passe_en_cours:
            mot_de_passe_en_cours = mot_de_passe_en_cours[:-1]
            # Jouer un bip grave court pour indiquer l'effacement
            jouer_bip(0.05, "G3")
            print(f"[DIGICODE] Effacement - Mot de passe actuel: ({mot_de_passe_en_cours})")
            # Si le mot de passe est vide, désactiver le mode saisie
            if not mot_de_passe_en_cours:
                saisie_mot_de_passe_active = False
        else:
            print("[DIGICODE] Aucun caractère à effacer")
            saisie_mot_de_passe_active = False
        # NE RIEN ENVOYER À L'ARDUINO - juste effacer localement
        return

    # 3. Validation du mot de passe avec '#'
    if touche == '#':
        if mot_de_passe_en_cours:
            print(f"[DIGICODE] Validation mot de passe détectée : {mot_de_passe_en_cours}")

            # Interroger le serveur pour authentifier le code de l'ouvrier
            try:
                reponse = requests.post(f"{URL_BASE}/authentifier", json={"password": mot_de_passe_en_cours}, timeout=2)
                if reponse.status_code == 200:
                    data = reponse.json()
                    if data.get("status") == "OK":
                        nom_ouvrier = data.get('nom', 'Inconnu')
                        envoyer_commande_arduino(arduino_serial, f"PASSWORD:{nom_ouvrier}")

                        # Jouer un son de succès pour l'authentification
                        jouer_bip_bon()

                        # Début de la phase de validation EPI
                        global etape_actuelle, dernier_ouvrier
                        etape_validation = True
                        etape_actuelle = 0
                        dernier_ouvrier = nom_ouvrier
                        saisie_mot_de_passe_active = False  # Désactiver la saisie
                        print(f"[VALIDATION] Début de la validation EPI pour {nom_ouvrier}")
                        print(f"[VALIDATION] Étape 1/{len(ETAPES_EPI)}: Vérifier le {ETAPES_EPI[0]['libelle']}")
                        envoyer_commande_arduino(arduino_serial, "DEMARRER_VALIDATION")
                    else:
                        envoyer_commande_arduino(arduino_serial, "BADGE:INCONNU")
                        # Jouer un son d'échec d'authentification
                        jouer_bip_erreur()
                        # Réinitialiser le mot de passe en cas d'échec
                        mot_de_passe_en_cours = ""
                        saisie_mot_de_passe_active = False
                else:
                    envoyer_commande_arduino(arduino_serial, "BADGE:INCONNU")
                    jouer_bip_erreur()
                    mot_de_passe_en_cours = ""
                    saisie_mot_de_passe_active = False
            except requests.exceptions.RequestException:
                print("[ERREUR] Serveur injoignable pour authentification")
                envoyer_commande_arduino(arduino_serial, "RESET")
                jouer_bip_erreur()
                mot_de_passe_en_cours = ""
                saisie_mot_de_passe_active = False

            mot_de_passe_en_cours = "" # Réinitialisation après envoi
        else:
            # Si on appuie sur # sans mot de passe, désactiver le mode saisie
            saisie_mot_de_passe_active = False
        return

    # 4. Touches 'A', 'B', 'C', 'D' sont ignorées silencieusement
    print(f"[DIGICODE] Touche ignorée: {touche}")

def thread_surveillance_digicode(arduino_serial):
    """Boucle de fond scrutant le digicode."""
    print("[MATÉRIEL] Surveillance du Digicode active.")
    while True:
        try:
            touche = lire_digicode()
            if touche is not None:
                print(f"[DIGICODE] Touche détectée : {touche}")
                gerer_action_bouton(arduino_serial, touche)
            time.sleep(0.05)
        except Exception as e:
            print(f"[ERREUR] Thread Digicode : {e}")
            time.sleep(0.5)

def surveiller_rfid(arduino_serial):
    """Boucle principale : Gère les scans de cartes RFID/EPI et de casques."""
    global etape_validation, saisie_mot_de_passe_active, uid_casque_actuel

    lecteur = SimpleMFRC522()
    print("[MATÉRIEL] Lecteur RFID actif. En attente de badge...")

    # Dictionnaire pour stocker l'UID des badges scannés récemment (éviter les doubles scans)
    derniers_scans = {}

    while True:
        try:
            # Vérifier si on est en mode validation ou en saisie de mot de passe
            if etape_validation or saisie_mot_de_passe_active:
                # Attendre un peu avant de vérifier à nouveau
                time.sleep(0.5)
                continue

            id_badge, _ = lecteur.read()
            uid_str = str(id_badge).strip()

            # Vérifier si c'est un double scan rapide
            temps_actuel = time.time()
            if uid_str in derniers_scans:
                if temps_actuel - derniers_scans[uid_str] < 2:  # Ignorer si moins de 2 secondes
                    print(f"[RFID] Double scan ignoré pour UID: {uid_str}")
                    time.sleep(0.5)
                    continue
            derniers_scans[uid_str] = temps_actuel

            print(f"\n[RFID] Badge détecté ! UID: {uid_str}")

            # Vérifier l'état de connexion pour cet UID
            connecte = est_connecte(uid_str)

            try:
                payload = {"uid": uid_str}
                reponse = requests.post(f"{URL_BASE}/badge", json=payload, timeout=2)

                if reponse.status_code == 200:
                    data = reponse.json()
                    role = str(data.get("role", "")).strip().upper()
                    nom = str(data.get("nom", "Inconnu")).strip()

                    if role == "ADMIN":
                        if not connecte:
                            # Premier scan - Connexion
                            envoyer_commande_arduino(arduino_serial, f"BADGE:ADMIN:{nom}")
                            basculer_etat(uid_str)
                            print(f"[RFID] ADMIN {nom} CONNECTÉ")
                            # Jouer un son de connexion admin
                            jouer_bip_bon()
                            # Envoyer un événement de connexion au backend
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "CONNEXION_ADMIN",
                                    "uid": uid_str,
                                    "nom": nom
                                }, timeout=1)
                            except:
                                pass
                        else:
                            # Deuxième scan - Déconnexion
                            envoyer_commande_arduino(arduino_serial, "BYE")
                            basculer_etat(uid_str)
                            print(f"[RFID] ADMIN {nom} DÉCONNECTÉ")
                            # Jouer un son de déconnexion (bip long grave)
                            jouer_bip(0.3, "G3")
                            # Envoyer un événement de déconnexion au backend
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "DECONNEXION_ADMIN",
                                    "uid": uid_str,
                                    "nom": nom
                                }, timeout=1)
                            except:
                                pass

                    elif role == "CASQUE":
                        if not connecte:
                            # Scan du casque - Démarrer le processus d'authentification
                            envoyer_commande_arduino(arduino_serial, "BADGE:CASQUE")
                            # Stocker l'UID du casque pour la validation
                            uid_casque_actuel = uid_str
                            print(f"[RFID] CASQUE DÉTECTÉ - Veuillez saisir votre mot de passe")
                            # Jouer un son pour signaler la détection du casque (arpeggio montant)
                            jouer_bip(0.05, "C4")
                            time.sleep(0.05)
                            jouer_bip(0.05, "E4")
                            time.sleep(0.05)
                            jouer_bip(0.05, "G4")
                            # Envoyer un événement de détection du casque
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "DETECTION_CASQUE",
                                    "uid": uid_str
                                }, timeout=1)
                            except:
                                pass
                            # Le casque sera connecté uniquement après validation EPI réussie
                        else:
                            # Si le casque est déjà connecté, déconnexion
                            envoyer_commande_arduino(arduino_serial, "BYE")
                            basculer_etat(uid_str)
                            uid_casque_actuel = None
                            print(f"[RFID] CASQUE DÉCONNECTÉ")
                            # Jouer un son de déconnexion
                            jouer_bip(0.3, "G3")
                            # Envoyer un événement de déconnexion
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "DECONNEXION_CASQUE",
                                    "uid": uid_str
                                }, timeout=1)
                            except:
                                pass

                    elif role == "OUVRIER":
                        if not connecte:
                            # Premier scan - Connexion
                            envoyer_commande_arduino(arduino_serial, f"BADGE:OUVRIER:{nom}")
                            basculer_etat(uid_str)
                            print(f"[RFID] OUVRIER {nom} CONNECTÉ")
                            # Jouer un son de connexion ouvrier
                            jouer_bip_bon()
                            # Envoyer un événement de connexion
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "CONNEXION_OUVRIER",
                                    "uid": uid_str,
                                    "nom": nom
                                }, timeout=1)
                            except:
                                pass
                        else:
                            # Deuxième scan - Déconnexion
                            envoyer_commande_arduino(arduino_serial, "BYE")
                            basculer_etat(uid_str)
                            print(f"[RFID] OUVRIER {nom} DÉCONNECTÉ")
                            # Jouer un son de déconnexion
                            jouer_bip(0.3, "G3")
                            # Envoyer un événement de déconnexion
                            try:
                                requests.post(f"{URL_BASE}/evenement", json={
                                    "action": "DECONNEXION_OUVRIER",
                                    "uid": uid_str,
                                    "nom": nom
                                }, timeout=1)
                            except:
                                pass
                    else:
                        envoyer_commande_arduino(arduino_serial, "BADGE:INCONNU")
                        # Jouer un son d'erreur pour badge inconnu
                        jouer_bip_erreur()
                else:
                    envoyer_commande_arduino(arduino_serial, "BADGE:INCONNU")
                    jouer_bip_erreur()

            except requests.exceptions.RequestException as e:
                print(f"[ERREUR] Connexion backend échouée : {e}")
                envoyer_commande_arduino(arduino_serial, "RESET")
                jouer_bip_erreur()

            time.sleep(1)

        except Exception as e:
            print(f"[ERREUR] Système RFID: {e}")
            time.sleep(0.5)

def main():
    print("[SYSTÈME] Démarrage du système unifié (RFID + Digicode + Buzzer)...")
    print("[BUZZER] Buzzer KXG1205 initialisé sur le GPIO 17")

    # Init commune de l'Arduino et de la structure GPIO
    arduino = initialiser_materiel()

    # Lancement du scanner de clavier en arrière-plan
    t_digicode = threading.Thread(target=thread_surveillance_digicode, args=(arduino,), daemon=True)
    t_digicode.start()

    # Lancement de la boucle RFID sur le thread principal
    try:
        surveiller_rfid(arduino)
    except KeyboardInterrupt:
        print("\n[SYSTÈME] Signal d'arrêt reçu.")
    finally:
        if arduino:
            arduino.close()
        # Nettoyer le buzzer
        buzzer.stop()
        GPIO.cleanup()
        print("[SYSTÈME] GPIO nettoyés. Système éteint.")

if __name__ == "__main__":
    main()

    "273425030702": { nom: "Sécurité Casque", role: "CASQUE" },
    "5555555555": { nom: "Pierre Martin", role: "OUVRIER", password: "1234" } // Ouvrier avec mot de passe
};

let historiquePassages = [];

/**
 * 1. Endpoint RFID
 */
app.post('/api/badge', (req, res) => {
    const { uid } = req.body;
    console.log(`\n[RFID] Scan UID : ${uid}`);

    const employe = BASE_EMPLOYES[uid];
    if (employe) {
        historiquePassages.push({ type: "RFID_SCAN", nom: employe.nom, role: employe.role, date: new Date() });
        return res.json({ role: employe.role, nom: employe.nom });
    } else {
        return res.json({ role: "INCONNU", nom: "Inconnu" });
    }
});

/**
 * 2. Endpoint Authentification par Digicode (Déclenché par le '#' sur le Pi)
 */
app.post('/api/authentifier', (req, res) => {
    const { password } = req.body;
    console.log(`[DIGICODE] Requête de vérification du code : ${password}`);

    // Trouver l'employé correspondant au mot de passe saisi
    const employe = Object.values(BASE_EMPLOYES).find(emp => emp.password === password);

    if (employe) {
        console.log(`[DIGICODE] Code Correct ! Ouvrier trouvé : ${employe.nom}`);
        historiquePassages.push({ type: "DIGICODE_AUTH", nom: employe.nom, role: employe.role, date: new Date() });
        return res.json({ status: "OK", nom: employe.nom });
    } else {
        console.log(`[DIGICODE] Code Incorrect.`);
        return res.json({ status: "ERREUR" });
    }
});

/**
 * 3. Endpoint Événements (Pour enregistrer les actions EPI reçues de vos touches A, B, C, D)
 */
app.post('/api/evenement', (req, res) => {
    const { action } = req.body;
    console.log(`[LOG MATÉRIEL] Commande reçue : ${action}`);

    historiquePassages.push({
        type: "EPI_NAVIGATION",
        action: action,
        date: new Date()
    });

    res.json({ status: "OK" });
});

/**
 * 4. Historique (React Native)
 */
app.get('/api/historique', (req, res) => {
    res.json(historiquePassages);
});

/**
 * 5. Ajouter un employé (React Native)
 */
app.post('/api/employes/ajouter', (req, res) => {
    const { uid, nom, role, password } = req.body;
    if (!uid || !nom) return res.status(400).json({ error: "Champs manquants" });

    BASE_EMPLOYES[uid] = {
        nom: nom,
        role: role ? role.toUpperCase() : "OUVRIER",
        password: password || null
    };
    res.json({ status: "success", message: "Employé configuré" });
});

    console.log(`==================================================`);
    console.log(`   SERVEUR BACKEND SÉCURISÉ ACTIF SUR LE PORT ${PORT}`);
    console.log(`==================================================`);
});
